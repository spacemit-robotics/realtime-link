# Native compiler defaults may target a newer CPU than the deployment board.
# Use the common RISC-V 64-bit Linux ABI by default; target-specific builds
# may explicitly disable this option and supply their own toolchain flags.
option(VIDEO_TRANSMISSION_PORTABLE_RISCV "Build for the common RV64GC/LP64D baseline" ON)
if(VIDEO_TRANSMISSION_PORTABLE_RISCV AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(riscv64|RISCV64)$")
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "Portable RISC-V build requires a GNU or Clang-compatible compiler")
 endif()
 include(CheckCXXCompilerFlag)
 check_cxx_compiler_flag("-march=rv64gc -mabi=lp64d" VIDEO_TRANSMISSION_HAS_RV64GC)
 if(NOT VIDEO_TRANSMISSION_HAS_RV64GC)
  message(FATAL_ERROR "Compiler does not support the RV64GC/LP64D baseline; check the toolchain")
 endif()
 add_compile_options(-march=rv64gc -mabi=lp64d)
 add_link_options(-march=rv64gc -mabi=lp64d)
endif()

# same installation and do not silently fall back to a host-system MPP.

 unset(_VT_MPP_PREFIX_INCLUDE CACHE)
 unset(_VT_MPP_PREFIX_LIBRARY CACHE)

 if(NOT _VT_MPP_PREFIX_INCLUDE OR NOT _VT_MPP_PREFIX_LIBRARY)
  message(FATAL_ERROR "VIDEO_TRANSMISSION_MPP_PREFIX must contain MPP headers and libmpp")
 endif()
 set(VT_MPP_BRIDGE_INCLUDE_DIR "${_VT_MPP_PREFIX_INCLUDE}")
 foreach(_vt_component CAMERA MEDIA MJPEG V2D)
  set(VT_MPP_${_vt_component}_LIBRARY "${_VT_MPP_PREFIX_LIBRARY}")
 endforeach()
 message(STATUS "MPP headers: ${_VT_MPP_PREFIX_INCLUDE}; library: ${_VT_MPP_PREFIX_LIBRARY}")
endif()
