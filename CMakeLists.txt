cmake_minimum_required(VERSION 3.16)
set(CMAKE_INSTALL_RPATH "$ORIGIN")
project(video_transmission VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)
option(FPV_BUILD_PIPELINE "Build the MPP pipeline and facade" ON)
option(FPV_BUILD_EXAMPLES "Build native FPV examples" ON)
option(FPV_TEST_PACED_LOOPBACK "Register timing-sensitive paced UDP integration tests" OFF)
add_library(video_transmission SHARED src/latency.cpp src/version.cpp src/protocol/legacy_udp.cpp src/sky/session.cpp src/ground/session.cpp src/transport/socket.cpp src/transport/usb.cpp src/config.cpp src/control/control.cpp)
add_library(video_transmission::video_transmission ALIAS video_transmission)
target_compile_features(video_transmission PUBLIC cxx_std_20)
target_compile_options(video_transmission PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(video_transmission PRIVATE src PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(video_transmission PRIVATE Threads::Threads yaml-cpp)
install(TARGETS video_transmission EXPORT coreTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/video_transmission/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/video_transmission
 PATTERN "fpv_session.hpp" EXCLUDE
 PATTERN "mpp_frame.hpp" EXCLUDE
 PATTERN "mjpeg_decoder.hpp" EXCLUDE)
install(EXPORT coreTargets FILE coreTargets.cmake NAMESPACE video_transmission:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/video_transmission)
if(FPV_BUILD_PIPELINE)
 find_package(mpp CONFIG REQUIRED)
 add_library(fpv_pipeline SHARED src/pipeline/session.cpp src/pipeline/mpp_frame.cpp)
 add_library(video_transmission::fpv_pipeline ALIAS fpv_pipeline)
 target_compile_features(fpv_pipeline PUBLIC cxx_std_20)
 target_include_directories(fpv_pipeline PRIVATE src PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
 target_link_libraries(fpv_pipeline PUBLIC video_transmission mpp::mpp PRIVATE Threads::Threads)
 install(FILES include/video_transmission/fpv_session.hpp include/video_transmission/mpp_frame.hpp
  include/video_transmission/mjpeg_decoder.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/video_transmission)
 install(TARGETS fpv_pipeline EXPORT pipelineTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
 install(EXPORT pipelineTargets FILE pipelineTargets.cmake NAMESPACE video_transmission:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/video_transmission)
endif()
if(FPV_BUILD_EXAMPLES)
 if(NOT FPV_BUILD_PIPELINE)
  message(FATAL_ERROR "FPV_BUILD_EXAMPLES requires FPV_BUILD_PIPELINE")
 endif()
 add_subdirectory(example)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake.in" [=[
@PACKAGE_INIT@
include("${CMAKE_CURRENT_LIST_DIR}/coreTargets.cmake")
foreach(component IN LISTS video_transmission_FIND_COMPONENTS)
 if(component STREQUAL "pipeline" AND EXISTS "${CMAKE_CURRENT_LIST_DIR}/pipelineTargets.cmake")
  include(CMakeFindDependencyMacro)
  find_dependency(mpp CONFIG)
  include("${CMAKE_CURRENT_LIST_DIR}/pipelineTargets.cmake")
  set(video_transmission_pipeline_FOUND TRUE)
 else()
  set(video_transmission_${component}_FOUND FALSE)
 endif()
endforeach()
check_required_components(video_transmission)
]=])
configure_package_config_file("${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake.in" ${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/video_transmission)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/video_transmission)
include(CTest)
if(BUILD_TESTING)
 add_executable(qp_config_test tests/qp_config_test.cpp)
 target_compile_options(qp_config_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
 target_link_libraries(qp_config_test PRIVATE video_transmission)
 add_test(NAME qp_config_test COMMAND qp_config_test)
 set_tests_properties(qp_config_test PROPERTIES TIMEOUT 5)
 add_executable(frame_timing_test tests/frame_timing_test.cpp)
 target_include_directories(frame_timing_test PRIVATE src)
 target_compile_options(frame_timing_test PRIVATE -UNDEBUG)
 target_link_libraries(frame_timing_test PRIVATE video_transmission Threads::Threads)
 add_test(NAME frame_timing_policy_test COMMAND frame_timing_test --policy-only)
 set_tests_properties(frame_timing_policy_test PROPERTIES TIMEOUT 5)
 if(FPV_TEST_PACED_LOOPBACK)
  add_test(NAME frame_timing_test COMMAND frame_timing_test)
  set_tests_properties(frame_timing_test PROPERTIES TIMEOUT 15 LABELS "timing-sensitive")
 endif()
 add_executable(video_transmission_test tests/session_test.cpp)
 target_include_directories(video_transmission_test PRIVATE src/protocol)
 target_link_libraries(video_transmission_test PRIVATE video_transmission)
 target_compile_options(video_transmission_test PRIVATE -UNDEBUG)
 add_test(NAME video_transmission_test COMMAND video_transmission_test)
 add_executable(config_control_test tests/config_control_test.cpp)
 target_compile_options(config_control_test PRIVATE -UNDEBUG)
 target_link_libraries(config_control_test PRIVATE video_transmission)
 set(FPV_TEST_CONFIG_DIR "${CMAKE_CURRENT_SOURCE_DIR}/example/config")
 add_test(NAME config_control_test COMMAND config_control_test ${FPV_TEST_CONFIG_DIR})
 if(FPV_BUILD_PIPELINE)
  add_executable(raw_encoder_test tests/raw_encoder_test.cpp src/pipeline/mpp_frame.cpp)
  target_compile_features(raw_encoder_test PRIVATE cxx_std_20)
  target_compile_options(raw_encoder_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  target_include_directories(raw_encoder_test PRIVATE src)
  target_link_libraries(raw_encoder_test PRIVATE video_transmission mpp::mpp Threads::Threads)
  add_test(NAME raw_encoder_test COMMAND raw_encoder_test ${FPV_TEST_CONFIG_DIR})
  set_tests_properties(raw_encoder_test PROPERTIES TIMEOUT 5)
  add_executable(facade_test tests/facade_test.cpp)
  target_compile_options(facade_test PRIVATE -UNDEBUG)
  target_link_libraries(facade_test PRIVATE fpv_pipeline)
  add_test(NAME facade_test COMMAND facade_test ${FPV_TEST_CONFIG_DIR})
 endif()
 set_tests_properties(video_transmission_test config_control_test PROPERTIES TIMEOUT 15)
 if(FPV_BUILD_PIPELINE)
  set_tests_properties(facade_test PROPERTIES TIMEOUT 15)
 endif()
 add_executable(usb_transport_test tests/usb_transport_test.cpp)
 target_compile_options(usb_transport_test PRIVATE -UNDEBUG)
 target_link_libraries(usb_transport_test PRIVATE video_transmission)
 add_test(NAME usb_transport_test COMMAND usb_transport_test)
 set_tests_properties(usb_transport_test PROPERTIES TIMEOUT 10)
endif()
