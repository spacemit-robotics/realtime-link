cmake_minimum_required(VERSION 3.16)
set(CMAKE_INSTALL_RPATH "$ORIGIN")
project(video_transmission VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
option(VIDEO_TRANSMISSION_BUILD_LEGACY "Build the internal product compatibility implementation" OFF)
option(VIDEO_TRANSMISSION_BUILD_PIPELINE "Build the internal MPP pipeline and facade" OFF)
option(VIDEO_TRANSMISSION_BUILD_EXAMPLES "Build native video transmission examples" ON)
option(VIDEO_TRANSMISSION_INSTALL_EXAMPLES "Install reference example programs and configuration" OFF)
option(VIDEO_TRANSMISSION_TEST_PACED_LOOPBACK "Register timing-sensitive paced UDP integration tests" OFF)
add_library(video_transmission SHARED src/public/session.cpp src/public/reference_udp.cpp)
add_library(video_transmission::video_transmission ALIAS video_transmission)
target_compile_features(video_transmission PUBLIC cxx_std_20)
target_compile_options(video_transmission PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(video_transmission PRIVATE VT_BUILDING_LIBRARY=1)
set_target_properties(video_transmission PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 target_link_options(video_transmission PRIVATE
  "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/public/exports.map")
endif()
target_include_directories(video_transmission
 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>
 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(video_transmission PRIVATE Threads::Threads)
install(TARGETS video_transmission EXPORT coreTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/video_transmission/video_transmission.hpp
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/video_transmission)
install(FILES README.md LICENSE DESTINATION share/video_transmission)
install(EXPORT coreTargets FILE coreTargets.cmake NAMESPACE video_transmission:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/video_transmission)
if(VIDEO_TRANSMISSION_BUILD_PIPELINE AND NOT VIDEO_TRANSMISSION_BUILD_LEGACY)
 message(FATAL_ERROR "VIDEO_TRANSMISSION_BUILD_PIPELINE requires VIDEO_TRANSMISSION_BUILD_LEGACY")
endif()
if(VIDEO_TRANSMISSION_BUILD_LEGACY)
 find_package(yaml-cpp REQUIRED)
 add_library(video_transmission_legacy SHARED src/latency.cpp src/version.cpp src/protocol/legacy_udp.cpp src/sky/session.cpp src/ground/session.cpp src/transport/socket.cpp src/transport/usb.cpp src/config.cpp src/control/control.cpp)
 target_compile_features(video_transmission_legacy PUBLIC cxx_std_20)
 target_compile_options(video_transmission_legacy PRIVATE -Wall -Wextra -Wpedantic)
 target_include_directories(video_transmission_legacy PRIVATE src PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
 target_link_libraries(video_transmission_legacy PRIVATE Threads::Threads yaml-cpp)
 if(VIDEO_TRANSMISSION_INSTALL_EXAMPLES)
  install(TARGETS video_transmission_legacy LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
 endif()
endif()
if(VIDEO_TRANSMISSION_BUILD_PIPELINE)
 find_package(mpp CONFIG QUIET)
 if(NOT TARGET mpp::mpp)
  find_path(VT_MPP_INCLUDE_DIR NAMES sys/sys_api.h)
  find_library(VT_MPP_LIBRARY NAMES mpp)
  if(NOT VT_MPP_INCLUDE_DIR OR NOT VT_MPP_LIBRARY)
   message(FATAL_ERROR "MPP headers and libmpp are required for the native pipeline")
  endif()
  set(VT_MPP_INCLUDE_DIRS "${VT_MPP_INCLUDE_DIR}")
  foreach(component sys vi vo venc vdec demux mux v2d uvc)
   if(IS_DIRECTORY "${VT_MPP_INCLUDE_DIR}/${component}")
    list(APPEND VT_MPP_INCLUDE_DIRS "${VT_MPP_INCLUDE_DIR}/${component}")
   endif()
  endforeach()
  add_library(mpp::mpp SHARED IMPORTED)
  set_target_properties(mpp::mpp PROPERTIES
   IMPORTED_LOCATION "${VT_MPP_LIBRARY}"
   INTERFACE_INCLUDE_DIRECTORIES "${VT_MPP_INCLUDE_DIRS}")
 endif()
 add_library(video_transmission_pipeline SHARED src/pipeline/session.cpp src/pipeline/mpp_frame.cpp)
 add_library(video_transmission::pipeline ALIAS video_transmission_pipeline)
 set_target_properties(video_transmission_pipeline PROPERTIES EXPORT_NAME pipeline)
 target_compile_features(video_transmission_pipeline PUBLIC cxx_std_20)
 target_include_directories(video_transmission_pipeline PRIVATE src PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
 target_link_libraries(video_transmission_pipeline PUBLIC video_transmission_legacy mpp::mpp PRIVATE Threads::Threads)
 target_compile_definitions(video_transmission_pipeline PUBLIC VIDEO_TRANSMISSION_HAS_PIPELINE=1)
 if(VIDEO_TRANSMISSION_INSTALL_EXAMPLES)
  install(TARGETS video_transmission_pipeline LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
 endif()
endif()
if(VIDEO_TRANSMISSION_BUILD_EXAMPLES)
 add_subdirectory(example)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake.in" [=[
@PACKAGE_INIT@
include("${CMAKE_CURRENT_LIST_DIR}/coreTargets.cmake")
check_required_components(video_transmission)
]=])
configure_package_config_file("${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake.in" ${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/video_transmission)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/video_transmissionConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/video_transmission)

  add_test(NAME frame_timing_test COMMAND frame_timing_test)
  set_tests_properties(frame_timing_test PROPERTIES TIMEOUT 15 LABELS "timing-sensitive")
 endif()

 target_include_directories(video_transmission_test PRIVATE src/protocol)

 add_test(NAME config_control_test COMMAND config_control_test ${VIDEO_TRANSMISSION_TEST_CONFIG_DIR})

  add_executable(raw_encoder_test tests/raw_encoder_test.cpp src/pipeline/mpp_frame.cpp)

 endif()
 set_tests_properties(video_transmission_test config_control_test PROPERTIES TIMEOUT 15)
 if(VIDEO_TRANSMISSION_BUILD_PIPELINE)
  set_tests_properties(facade_test PROPERTIES TIMEOUT 15)
 endif()
 add_executable(usb_transport_test tests/usb_transport_test.cpp)
 target_compile_options(usb_transport_test PRIVATE -UNDEBUG)
